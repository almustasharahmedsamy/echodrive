cmake_minimum_required(VERSION 3.20)
project(EchoDrive VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(echodrive src/main.cpp src/game.cpp src/voice.cpp src/settings.cpp src/content.cpp src/spatial_audio.cpp src/audio_runtime.cpp src/audio_assets.cpp src/miniaudio_impl.cpp)
target_include_directories(echodrive PRIVATE include third_party)
if(WIN32)
  target_compile_definitions(echodrive PRIVATE ECHODRIVE_WINDOWS=1 UNICODE _UNICODE)
  target_link_libraries(echodrive PRIVATE ole32 sapi winmm)
endif()
if(MSVC)
  target_compile_options(echodrive PRIVATE /W4 /permissive-)
else()
  target_compile_options(echodrive PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS echodrive RUNTIME DESTINATION bin)
